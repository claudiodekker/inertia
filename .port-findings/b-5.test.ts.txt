// PR: layerPageOf "falls back to the address for a layer with no url of its own": the layer's page.url is the address
//   beneath it ('/users').
// Rebuilt: a local layer's page.url is '' (stack.ts openLocal), so usePage().url inside it is '' not '/users'.
import { describe, expect, it } from 'vitest'
import { page, ported, settled } from '../packages/core/tests/layers/ported/support'

describe('layerPageOf', () => {
  it('falls back to the address for a layer with no url of its own', async () => {
    const app = await ported(page('Users/Index', '/users', { props: { errors: {}, users: [] } }), {})

    app.router.layer({ component: 'Prompt' })
    await settled()

    expect(app.views()[0].page.url).toBe('/users')
  })
})
