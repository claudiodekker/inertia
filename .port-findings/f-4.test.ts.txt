// FINDING f-4. PR: a throwing tooling handler never breaks the write that announced it.
// Rebuilt: layerSlot.subscribe is the only observation point; a throwing listener rejects Painter.paint, the layer
// still opens (layers: 1, history pushed) but the visit fires inertia:networkError instead of inertia:success,
// skips inertia:navigate and onSuccess, and leaves an unhandled rejection.
import { expect, it } from 'vitest'
import { page, settled, start } from './f-support'

const users = page('Users/Index', '/users', { props: { errors: {}, users: [] } })
const edit = page('Users/Edit', '/users/5/edit', { layer: { key: 'Users/Edit', base: '/users' } })

it('keeps a handler that throws from breaking the write that announced', async () => {
  const app = await start(users, { '/users/5/edit': edit })
  const events: string[] = []

  app.subscribe(() => {
    throw new Error('tooling bug')
  })
  app.router.visit('/users/5/edit', { onSuccess: () => events.push('success'), onFinish: () => events.push('finish') })
  await settled()

  expect(app.layers().map((layer) => layer.component)).toEqual(['Users/Edit'])
  expect(events).toEqual(['success', 'finish'])
})
