import { expect, it } from 'vitest'
import { bench, byPath, gate, page, settled } from './e-support'

it('inserts a base that is itself a layer beneath the stack and keeps walking (loading base url)', async () => {
  const walk = gate()
  const edit = page('Users/Edit', '/users/5/edit', { layer: { key: 'Users/Edit', base: '/users' } })
  const app = await bench(
    page('Auth/Login', '/login'),
    byPath({
      '/login': edit,
      '/users': page('Users/Index', '/users', { layer: { key: 'Users/Index', base: '/' } }),
      '/': () => walk.opened.then(() => page('Home', '/')),
    }),
  )

  app.router.visit('/login')
  await settled()

  expect(await app.basePage()).toMatchObject({ component: '', url: '/' })
})
