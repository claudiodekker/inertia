// PR: layerPageOf "falls back to the address beneath it, never the address of a layer above it":
//   local layer page.url === '/users' while the routed layer above it has '/users/9/notes'.
// Rebuilt: the local layer's page.url is '' (the routed layer above keeps '/users/9/notes').
import { describe, expect, it } from 'vitest'
import { page, ported, settled } from '../packages/core/tests/layers/ported/support'

describe('layerPageOf', () => {
  it('falls back to the address beneath it, never the address of a layer above it', async () => {
    const notes = page('Users/Notes', '/users/9/notes', { layer: { key: 'notes', base: '/users' } })
    const app = await ported(page('Users/Index', '/users', { props: { errors: {}, users: [] } }), {
      '/users/9/notes': notes,
    })

    app.router.layer({ component: 'Prompt' })
    await settled()
    app.router.visit('/users/9/notes')
    await settled()

    expect(app.views().map((view) => view.page.url)).toEqual(['/users', '/users/9/notes'])
  })
})
