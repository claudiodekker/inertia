// FINDING c-5 (PR row 1747): PR outcome: closing the standalone bottom layer of [standalone, notes (1 entry), teams (1 entry)]
// calls history.go(-2) and the close does not resolve until the browser answers.
// Rebuilt outcome: no history.go at all (close.ts removeOrUnwind: `covered[0].standalone || entries === 0` pushes the base instead),
// written ends with ['push','/users'], so the two entries pushed over the cold layer stay in session history.
// Uses holdBack/ported/layerAt from layer-handles-support; routes: /users/5/edit, /users/5/notes, /teams/9 as in layer-handles.test.ts 'closing a layer'
describe('finding c-5', () => {
  it('closes every layer above the one it was asked to close', async () => {
    const app = await ported(base(), routes())
    const cold = app.router.layer('http://localhost/users/5/edit', { component: 'Users/Loading' } as never)

    await settled()
    app.router.layer('http://localhost/users/5/notes')
    await settled()
    app.router.layer('http://localhost/teams/9')
    await settled()

    const browser = holdBack()
    let resolved = false
    const closing = cold.close().then(() => (resolved = true))

    await settled()
    app.exitShells()
    await settled()

    expect(browser.held).toEqual([-2])
    expect(resolved).toBe(false)
    expect(app.layers()).toHaveLength(3)

    browser.answer()
    await closing

    expect(app.layers()).toEqual([])
  })

})
