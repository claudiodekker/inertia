// FINDING a2-3: a partial answered by the real base while a cold layer's walk is in flight drops the layer instead of keeping it.
// PR (layer-responses.test.ts:1370): base becomes Users/Index, layers stay [layer.id], base props contain stats [1, 2, 3].
// Rebuilt: visit('/users', {only: ['stats'], async: true}) answered by pw({props: {users: [{id: 9}], stats: [1, 2, 3]}}): base is Users/Index with
//   those props (onSuccess fires) but app.layers() === []: the replacement of the blank base is an ordinary base commit and drops every row;
//   only the walk's own landing keeps them.
// Needs the helpers of layer-responses-2.test.ts (pw, editLayer, visited, scenario, held, settled).

  const cold = async (layerProps: Record<string, unknown> = { user: { id: 5 } }) => {
    const walk = held()
    let hops = 0
    const routes: Record<string, Route> = {
      'post /login': editLayer({ layer: { key: 'Users/Edit', base: '/users' }, props: layerProps }),
      '/users': (request) => (hops++ === 0 ? walk.answer() : Promise.resolve(pw())),
    }
    const app = await scenario(pw({ component: 'Auth/Login', url: '/login' }), routes)

    await visited(app, '/login', { method: 'post' })
    await settled()

    expect((await app.basePage()).component).toBe('')

    return { app, routes, walk, layer: app.layers()[0] }
  }

  it('replaces the blank with the real base when a partial is answered by the base, keeping the layer', async () => {
    const { app, routes, layer } = await cold({ user: { id: 5 } })

    routes['/users'] = pw({ props: { users: [{ id: 9 }], stats: [1, 2, 3] } })
    await visited(app, '/users', { only: ['stats'], async: true })
    await settled()

    expect((await app.basePage()).component).toBe('Users/Index')
    expect(app.layers().map((open) => open.id)).toEqual([layer.id])
    expect((await app.basePage()).props).toMatchObject({ stats: [1, 2, 3] })
  })
