// PR: history entry counting "counts the entry against the layer that owns the address": an in-page anchor entry
//   (popstate with a null state) is recorded on the top layer (recordHistoryEntry) so closing unwinds it too:
//   after open (1 entry) + anchor, layer.entries === 2 and entriesToUnwind === 2.
// Rebuilt: the null-state branch (eventHandler.ts handlePopstateEvent) never touches layer rows: entries stays 1 after the
//   anchor, closing issues history.go(-1) and leaves the anchor entry above the base (location 'http://localhost/users#top').
import { describe, expect, it } from 'vitest'
import { page, ported, settled } from '../packages/core/tests/layers/ported/support'

describe('history entry counting', () => {
  it('counts the entry against the layer that owns the address', async () => {
    const app = await ported(page('Users/Index', '/users', { props: { errors: {}, users: [] } }), {
      '/a': page('A', '/a', { layer: { key: 'A', base: '/users' } }),
    })

    app.router.visit('/a')
    await settled()
    await app.anchor('/a#top')

    const deltas: number[] = []

    window.history.go = (delta: number) => void deltas.push(delta)

    expect(app.entry().layers![0].entries).toBe(2)

    void app.router.close()
    await settled()
    app.exitShells()
    await settled()

    expect(deltas).toEqual([-2])
  })
})
