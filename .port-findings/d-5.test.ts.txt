// (same root cause as d-1)
// PR outcome: after the user visits '/dashboard', the stale hop for '/users' is answered with 409 + x-inertia-location + new version; 3.x still processes that
//   answer: exactly one 'inertia:location' event is dispatched, base stays Dashboard, no layers, router.visit called once (no recovery visit).
// Rebuilt outcome: base Dashboard, layers [], router.visit called once (all hold), but 0 'inertia:location' events: the user's router.visit('/dashboard') cancels the
//   in-flight hop to '/users' (3.x cancels async requests aimed at the page being navigated away from; the blank base url is the hop's url), so its 409 is never processed.
it('does not re-request the blank when the user has navigated away from the stack being walked', async () => {
  const w = await world({
    ...from(),
    routes: { '/users': 'hold', '/dashboard': pageWith({ component: 'Dashboard', url: '/dashboard' }) },
  })
  await w.begin(login)
  await w.open(layerAt('/users/5/edit', 'Users/Edit', '/users'))
  const visit = vi.spyOn(w.router, 'visit')

  w.router.visit('/dashboard')
  await settled()
  await w.answer('/users', elsewhere)

  expect(w.events.filter((type) => type === 'inertia:location')).toHaveLength(1) // FAILS: 0
  expect((await w.base()).component).toBe('Dashboard')
  expect(w.layers()).toEqual([])
  expect(visit).toHaveBeenCalledTimes(1)
})
