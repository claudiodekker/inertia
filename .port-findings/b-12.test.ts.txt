// PR: history entry counting "gives the entry back when the browser refused to write it": a pushState the browser refuses
//   (QuotaExceededError) leaves the layer with 0 entries, so closing it unwinds 0 entries.
// Rebuilt: the layer commit is charged one entry when the push is decided, before the write; the refused push (3.x then
//   reloads the document to the url) is never given back, so closing the layer issues history.go(-1).
import { describe, expect, it } from 'vitest'
import { page, ported, settled } from '../packages/core/tests/layers/ported/support'

describe('history entry counting', () => {
  it('gives the entry back when the browser refused to write it', async () => {
    const app = await ported(page('Users/Index', '/users', { props: { errors: {}, users: [] } }), {
      '/a': page('A', '/a', { layer: { key: 'A', base: '/users' } }),
    })
    const reloads: string[] = []
    const pushState = window.history.pushState

    Object.defineProperty(window.location, 'href', {
      configurable: true,
      get: () => 'http://localhost/users',
      set: (value: string) => void reloads.push(value),
    })
    window.history.pushState = () => {
      const error = new Error('quota')
      error.name = 'QuotaExceededError'
      throw error
    }

    app.router.visit('/a')
    await settled()
    window.history.pushState = pushState

    const deltas: number[] = []

    window.history.go = (delta: number) => void deltas.push(delta)
    void app.router.close()
    await settled()
    app.exitShells()
    await settled()

    expect(deltas).toEqual([])
  })
})
