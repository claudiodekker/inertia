// the base a layer response composes onto > takes the asset version the layer arrived on   (PR layer-responses.test.ts:689)
// PR: base page version 'old', a layer answer carrying version 'new' -> the page version becomes 'new'.
// Rebuilt: basePage().version stays 'old' (only the layer's own page has 'new'); the base's next requests still send
// X-Inertia-Version 'old' (requests: old, old, then 'new' only for the layer's own router).
import { expect, it } from 'vitest'
import { page, scenario, settled } from './a-support'

it('takes the asset version the layer arrived on', async () => {
  const app = await scenario(page('Users/Index', '/users', { version: 'old', props: { users: [], errors: {} } }), {
    'post /users/5/edit': page('Users/Edit', '/users/5/edit', { version: 'new', layer: { key: 'Users/Edit' } }),
  })

  app.router.post('/users/5/edit')
  await settled()

  expect((await app.basePage()).version).toBe('new')
})
