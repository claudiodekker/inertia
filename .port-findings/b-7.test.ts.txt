// PR: resolveLayers "leaves the url empty for a layer with no url of its own, so it never reaches the address":
//   layer.layoutPage.url === '' AND layer.page.url === '/users' (the layer's own code still reads the address beneath it).
// Rebuilt: only one page per layer. For a local layer both are the same object: page.url is '' (the layoutPage half holds,
//   the "own code reads the address beneath" half does not): views()[0].page.url === ''.
import { describe, expect, it } from 'vitest'
import { page, ported, settled } from '../packages/core/tests/layers/ported/support'

describe('resolveLayers, the layout page', () => {
  it('leaves the url empty for a layer with no url of its own, so it never reaches the address', async () => {
    const app = await ported(page('Users/Index', '/users', { props: { errors: {}, users: [] } }), {})

    app.router.layer({ component: 'Users/Edit', props: { user: { id: 5 }, errors: { name: 'required' } } })
    await settled()

    expect(app.views()[0].page.url).toBe('')
    expect(app.views()[0].page.url).toBe('/users')
  })
})
