// FINDING: after the response keyed 'panel:second' lands on the fabricated placeholder, layers() info key stays 'panel:first' (PR: 'panel:second'); url/component/id/props land
// uses helpers from layer-responses-3.test.ts (index, edit, open, scopeOf, inertia, held, scenario)
  it('lands a response keyed differently on the placeholder rather than beside it', async () => {
    const { app, next } = await stood({ layer: { key: 'panel:first' } })
    const [layer] = app.layers()

    scopeOf(app).router.visit('/layers/panel/second', { component: 'Layers/Panel' })
    app.core.http.setClient({
      request: () => next.answer().then(inertia),
    })
    await settled()

    next.release(
      page('Layers/Panel', '/layers/panel/second', {
        layer: { key: 'panel:second' },
        props: { name: 'second', errors: {} },
      }),
    )
    await settled()

    expect(app.layers()).toEqual([
      { id: layer.id, key: 'panel:second', component: 'Layers/Panel', url: '/layers/panel/second', type: 'routed' },
    ])
    expect(app.views()[0].page.props).toEqual({ name: 'second', errors: {} })
  })
