// FINDING: after response: handle.state 'closed', base Users/Edit /users/5/edit (promoted), layers [] (PR: base Users/Index, one layer with the handle's id)
// uses helpers from layer-responses-3.test.ts (index, edit, open, scopeOf, inertia, held, scenario)
  it('lands the response on the placeholder an open put up, under the id it created', async () => {
    const next = held()
    const app = await scenario(index(), { '/users/5/edit': () => next.answer() })

    const handle = app.router.layer('/users/5/edit', { component: 'Users/Edit' })
    await settled()

    const fabricated = app.layers()[0]

    next.release(edit())
    await settled()

    expect((await app.basePage()).component).toBe('Users/Index')
    expect(app.layers()).toEqual([
      { id: fabricated.id, key: 'Users/Edit', component: 'Users/Edit', url: '/users/5/edit', type: 'routed' },
    ])
    expect(fabricated.id).toBe(handle.id)
    expect(app.views()[0].page.props).toEqual({ user: { id: 5 }, errors: {} })
  })
