// FINDING c-4 (PR row 1671 'counts a history entry an anchor pushed inside a layer'):
// PR outcome: an anchor pushing /users/5/edit#notes inside a layer (popstate with state null) makes layer entries [2]; closing then calls go(-2).
// Rebuilt outcome: layer entries stay [1] (written: push /users/5/edit#notes, then replace /users/5/edit); closing calls go(-1),
// so the user lands on the hash entry's predecessor, the layer's own open entry, instead of the page beneath.
// Helpers: ./layer-handles-support (ported, base, edit, layerAt, settled, holdBack)
import { describe, expect, it } from 'vitest'
import { base, edit, ported, settled } from './layer-handles-support'

describe('finding c-4', () => {
  it('counts a history entry an anchor pushed inside a layer', async () => {
    const app = await ported(base(), { '/users/5/edit': edit(), '/users': base() })
    const handle = app.router.layer('http://localhost/users/5/edit')
    await settled()

    window.history.pushState(null, '', '/users/5/edit#notes')
    window.history.go(0)
    await settled()

    expect((app.entry().layers as { entries: number }[]).map((layer) => layer.entries)).toEqual([2])

    const go = app.go()
    await app.closing(handle)

    expect(go).toEqual([-2])
  })
})
