// FINDING: local layer submits a POST answered by the layer beneath it (layer-marked Users/Edit page with errors at its own url): rebuilt classify rewrites the Users/Edit layer and dismisses the local layer (layers == [editId], local gone); PR kept both [editId, localId] with errors on the local layer only
// helpers openLocalOverALayer/submit as in layer-responses-3.test.ts
  it('keeps a local layer up when the address it was submitted from belongs to the layer beneath it', async () => {
    const { app, handle, editId } = await openLocalOverALayer({
      'post /users': edit({ props: { user: { id: 5 }, errors: { name: 'The name is required.' } } }),
    })

    const announced = await submit(app, handle.id)

    expect(announced).toEqual({ name: 'The name is required.' })
    expect(app.layers().map((layer) => layer.id)).toEqual([editId, handle.id])
    expect(app.views()[1].page.props.errors).toEqual({ name: 'The name is required.' })
    expect(app.views()[0].page.props.errors).toEqual({})
  })
